cmake_minimum_required(VERSION 3.24)
project(PulseWatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(PulseWatch WIN32 src/main.cpp)
target_compile_definitions(PulseWatch PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PulseWatch PRIVATE shell32 iphlpapi user32 gdi32 advapi32 d2d1 dwrite)
